export default { // 1. CHẠY TỰ ĐỘNG THEO CRON TRIGGER (6 LẦN / NGÀY) async scheduled(event, env, ctx) { ctx.waitUntil(updateAllVietlottDataToGithub(env)); }, // 2. XỬ LÝ YÊU CẦU HTTP TỪ APP async fetch(request, env, ctx) { const corsHeaders = { 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'GET, POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', }; if (request.method === 'OPTIONS') { return new Response(null, { headers: corsHeaders }); } const url = new URL(request.url); // Endpoint xác thực mật khẩu if (url.pathname === '/api/verify-password' && request.method === 'POST') { try { const body = await request.json(); if (body.password && body.password === env.APP_PASSWORD) { return new Response(JSON.stringify({ success: true, token: 'auth_session_' + Date.now(), message: 'Mật khẩu chính xác!' }), { status: 200, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } return new Response(JSON.stringify({ success: false, message: 'Mật khẩu không đúng!' }), { status: 401, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } catch (e) { return new Response(JSON.stringify({ success: false, message: 'Lỗi request!' }), { status: 400, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } } // Endpoint ép Worker chạy cập nhật thủ công (để test nhanh) if (url.pathname === '/api/trigger-sync' && request.method === 'POST') { const result = await updateAllVietlottDataToGithub(env); return new Response(JSON.stringify(result), { status: result.success ? 200 : 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }); } return new Response('Vietlott Worker & Sync Engine Active', { headers: corsHeaders }); } }; // Hàm cào dữ liệu Vietlott và Push lên GitHub async function updateAllVietlottDataToGithub(env) { const GITHUB_OWNER = "nguyenchemical"; const GITHUB_REPO = "Vietlott-Live-Gemini"; const GITHUB_BRANCH = "main"; // Lấy Secret GH_TOKEN đã cấu hình sẵn trong Cloudflare const TOKEN = env.GH_TOKEN; if (!TOKEN) { return { success: false, message: "Thiếu GH_TOKEN trong Secrets của Worker!" }; } const games = ['mega645', 'power655', 'lotto535', 'max3d', 'max3dplus', 'max3dpro']; const logResults = []; for (const game of games) { try { // Cào dữ liệu mới nhất const latestData = await fetchLatestDataFromVietlott(game); const filePath = `data/${game}.json`; const githubApiUrl = `https://api.github.com/repos/${GITHUB_OWNER}/${GITHUB_REPO}/contents/${filePath}?ref=${GITHUB_BRANCH}`; // 1. Lấy SHA của file hiện tại trên Repo let sha = ""; const getFileRes = await fetch(githubApiUrl, { headers: { 'Authorization': `token ${TOKEN}`, 'User-Agent': 'Cloudflare-Worker-Vietlott-Sync' } }); if (getFileRes.ok) { const fileInfo = await getFileRes.json(); sha = fileInfo.sha; } // 2. Commit & Push file JSON cập nhật lên GitHub const contentEncoded = btoa(unescape(encodeURIComponent(JSON.stringify(latestData, null, 2)))); const updateRes = await fetch(githubApiUrl, { method: 'PUT', headers: { 'Authorization': `token ${TOKEN}`, 'Content-Type': 'application/json', 'User-Agent': 'Cloudflare-Worker-Vietlott-Sync' }, body: JSON.stringify({ message: `[Auto Sync Worker] Cập nhật dữ liệu ${game} ngày ${new Date().toLocaleDateString('vi-VN')}`, content: contentEncoded, sha: sha || undefined, branch: GITHUB_BRANCH }) }); if (updateRes.ok) { logResults.push({ game, status: 'synced' }); } else { logResults.push({ game, status: 'failed_push', error: await updateRes.text() }); } } catch (err) { logResults.push({ game, status: 'error', error: err.message }); } } return { success: true, details: logResults }; } // Hàm giả lập cào dữ liệu từ Vietlott async function fetchLatestDataFromVietlott(gameType) { const getRandomInt = (min, max) => Math.floor(Math.random() * (max - min + 1)) + min; const draws = []; const todayStr = new Date().toLocaleDateString('vi-VN'); for (let i = 0; i < 30; i++) { if (gameType.startsWith('max3d')) { const count = gameType === 'max3d' ? 1 : 2; const sets = []; for (let k = 0; k < count; k++) sets.push(`${getRandomInt(0, 9)}${getRandomInt(0, 9)}${getRandomInt(0, 9)}`); draws.push({ id: `#00${1350 - i}`, date: todayStr, numbers: sets }); } else { const maxNum = gameType === 'mega645' ? 45 : (gameType === 'power655' ? 55 : 35); const set = new Set(); while (set.size < 6) set.add(getRandomInt(1, maxNum)); draws.push({ id: `#00${1350 - i}`, date: todayStr, numbers: Array.from(set).sort((a,b)=>a-b) }); } } return draws; }
